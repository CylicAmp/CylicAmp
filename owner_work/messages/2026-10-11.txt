### 6\. Reproducible intersection test

Here is a compact Python implementation of the construction and the exact geometric test. It uses rational arithmetic so intersections are not distorted by floating-point rounding.

```python

from fractions import Fraction as F
from itertools import combinations

# Cyclic order: A0,...,A5,B5,...,B0
V = [(2*i, 3) for i in range(6)]
V += [(2*i, -3) for i in range(5, -1, -1)]

# Circulant graph C_12(1,2,3,6)
E = set()
for i in range(12):
    for d in (1, 2, 3, 6):
        j = (i + d) % 12
        E.add(tuple(sorted((i, j))))

assert len(V) == 12
assert len(E) == 42
assert all(sum(i in e for e in E) == 7
           for i in range(12))

def sub(a, b):
    return (a[0] - b[0], a[1] - b[1])

def cross(a, b):
    return a[0]*b[1] - a[1]*b[0]

def proper_intersection(e, f):
    # Exclude shared graph vertices
    if set(e) & set(f):
        return None

    p, q = V[e[0]], V[e[1]]
    r, s = V[f[0]], V[f[1]]
    a, b = sub(q, p), sub(s, r)
    den = cross(a, b)

    # Parallel or collinear
    if den == 0:
        return None

    delta = sub(r, p)
    t = F(cross(delta, b), den)
    u = F(cross(delta, a), den)

    # Strictly inside both segments
    if not (0 < t < 1 and 0 < u < 1):
        return None

    return (F(p[0]) + t*a[0],
            F(p[1]) + t*a[1])

crossings = {}
for e, f in combinations(sorted(E), 2):
    p = proper_intersection(e, f)
    if p is not None:
        crossings.setdefault(p, []).append((e, f))

print("Vertices:", len(V))
print("Edges:", len(E))
print("Degree sequence:",
      [sum(i in e for e in E) for i in range(12)])
print("Proper crossing pairs:",
      sum(map(len, crossings.values())))
print("Unique crossing points:", len(crossings))
print("Central concurrency multiplicity:",
      len(crossings.get((F(5), F(0)), [])))
print("All unique intersections:")
for p in sorted(crossings):
    print(p, "pair count =", len(crossings[p]))
```
